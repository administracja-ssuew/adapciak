Append this IIFE immediately before `</script>`:

```javascript
/* ─── ANIM-02: Parallax logo ─── */
(function initParallax() {
  const hero = document.getElementById('start');
  if (!hero) return;
  const logo = hero.querySelector('.logo-blend');
  if (!logo) return;
  let ticking = false;

  function applyParallax() {
    const scrollY = window.scrollY;
    const heroH = hero.offsetHeight;
    if (scrollY > heroH) {
      ticking = false;
      return; // outside hero — skip transform
    }
    // Logo moves at 40% of scroll speed — adjust factor (0.3–0.4) if effect too strong
    logo.style.transform = 'translateY(' + (scrollY * 0.4) + 'px)';
    ticking = false;
  }

  window.addEventListener('scroll', function () {
    if (!ticking) {
      requestAnimationFrame(applyParallax);
      ticking = true;
    }
  }, { passive: true });
})();
```

Do NOT apply the transform to the wrapper div that has the `reveal` class. Apply only to the `<img class="logo-blend">` element, which is what `hero.querySelector('.logo-blend')` targets.
