```javascript
/* ─── FORM-01/02: Formspree contact form ─── */
/* <!-- WSTAW TU SWÓJ ENDPOINT FORMSPREE: zmień YOUR_FORM_ID na ID z formspree.io --> */
var FORMSPREE_URL = 'https://formspree.io/f/YOUR_FORM_ID';

async function handleFormSubmit(e) {
  e.preventDefault();
  var form    = e.target;
  var btn     = form.querySelector('button[type="submit"]');
  var success = document.getElementById('form-success');
  var error   = document.getElementById('form-error');

  if (success) success.classList.add('hidden');
  if (error)   error.classList.add('hidden');
  if (btn)     { btn.disabled = true; btn.textContent = 'Wysyłanie…'; }

  try {
    var res = await fetch(FORMSPREE_URL, {
      method:  'POST',
      headers: { 'Accept': 'application/json' },
      body:    new FormData(form)
    });

    if (res.ok) {
      form.reset();
      if (success) success.classList.remove('hidden');
    } else {
      var data = await res.json().catch(function() { return {}; });
      var msg  = (data.errors || []).map(function(err) { return err.message; }).join(', ');
      if (error) {
        error.textContent = msg || 'Wystąpił błąd. Spróbuj ponownie.';
        error.classList.remove('hidden');
      }
    }
  } catch (err) {
    if (error) {
      error.textContent = 'Błąd połączenia. Sprawdź internet i spróbuj ponownie.';
      error.classList.remove('hidden');
    }
  } finally {
    if (btn) { btn.disabled = false; btn.textContent = 'WYŚLIJ WIADOMOŚĆ'; }
  }
}
```
